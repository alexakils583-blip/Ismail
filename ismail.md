# step 1:Create a Next.js folder,
open the cmd terminal:
cd desktop 
npx create-next-app@latest ismail_nextjs
after packages install
cd ismail_next.js
npm install@supabase/supabase.js
npm run dev,
and then "https://localhost:3000"

# step 2:Install supabase,
open the google and type for supabase
login supabase with our google account
after login we create our project 
and connect supabase with folder next.js file .env.local

# step 3:Create the supabase file,
create file on next.js folder .env.local
after create file paste supabase link
NEXT_PUBLIC_SUPABASE_URL=your_supabase_url
NEXT_PUBLIC_SUPABASE_ANON_KEY=your_supabase_anon_key

# step 4:Create table note table,
go to supabase SQL editor and run
create table notes(
    id bigint generated by default as identity primary key,
    title text,
    content text,
    created_at timestamp with time zone default now()
);
Now your notes table is ready.
